#include <bits/stdc++.h>
using namespace std;

int main(){

    int n;

    cout<<"Enter number of processes: ";
    cin>>n;

    vector<int> AT(n), BT(n), REM(n),
                CT(n), TAT(n), WT(n), RT(n,-1);

    vector<bool> done(n,false);

    cout<<"\nEnter AT and BT:\n";

    for(int i=0;i<n;i++){

        cin>>AT[i]>>BT[i];

        REM[i]=BT[i];
    }

    int completed=0;
    int counter=0;

    cout<<"\nGantt Chart:\n";

    while(completed<n){

        int idx=-1;

        // SRTF -> minimum remaining time
        for(int i=0;i<n;i++){

            if(done[i]==false && AT[i]<=counter){

                if(idx==-1 ||
                   REM[i]<REM[idx] ||
                  (REM[i]==REM[idx] && AT[i]<AT[idx]) ||
                  (REM[i]==REM[idx] && AT[i]==AT[idx] && i<idx)){

                    idx=i;
                }
            }
        }

        // CPU idle
        if(idx==-1){

            cout<<"-1("<<counter<<")->";

            counter++;

            continue;
        }

        // response time = first cpu time - arrival time
        if(RT[idx]==-1)
            RT[idx]=counter-AT[idx];

        cout<<"P"<<idx<<"("<<counter<<")->";

        REM[idx]--;

        counter++;

        // process completed
        if(REM[idx]==0){

            CT[idx]=counter;

            TAT[idx]=CT[idx]-AT[idx];

            WT[idx]=TAT[idx]-BT[idx];

            done[idx]=true;

            completed++;
        }
    }

    cout<<"\n\nPID\tAT\tBT\tCT\tTAT\tWT\tRT\n";

    for(int i=0;i<n;i++){

        cout<<i<<"\t"
            <<AT[i]<<"\t"
            <<BT[i]<<"\t"
            <<CT[i]<<"\t"
            <<TAT[i]<<"\t"
            <<WT[i]<<"\t"
            <<RT[i]<<endl;
    }

    return 0;
}
