#include <bits/stdc++.h>
using namespace std;

int main(){

    int n;

    cout<<"Enter number of processes: ";
    cin>>n;

    vector<int> AT(n), BT(n), CT(n), TAT(n), WT(n), RT(n);

    vector<bool> done(n,false);

    cout<<"\nEnter AT and BT:\n";

    for(int i=0;i<n;i++)
        cin>>AT[i]>>BT[i];

    int completed=0;
    int counter=0;

    cout<<"\nGantt Chart:\n";

    while(completed<n){

        int idx=-1;

        // SJF -> minimum BT among arrived processes
        for(int i=0;i<n;i++){

            if(done[i]==false && AT[i]<=counter){

                if(idx==-1 ||
                   BT[i]<BT[idx] ||
                  (BT[i]==BT[idx] && AT[i]<AT[idx]) ||
                  (BT[i]==BT[idx] && AT[i]==AT[idx] && i<idx)){

                    idx=i;
                }
            }
        }

        // CPU idle
        if(idx==-1){

            cout<<"-1("<<counter<<")->";

            counter++;

            continue;
        }

        // your RT meaning = first cpu allocation time
        RT[idx]=counter-AT[idx]; //instead remember RT=WT in sjf-np imppppppppppppppppp or that RT=cpu first i.e left-At for all except fcfs

        cout<<"P"<<idx<<"("<<counter<<")->";

        counter+=BT[idx];

        CT[idx]=counter;

        TAT[idx]=CT[idx]-AT[idx];

        WT[idx]=TAT[idx]-BT[idx];

        done[idx]=true;

        completed++;
    }

    cout<<"\n\nPID\tAT\tBT\tCT\tTAT\tWT\tRT\n";

    for(int i=0;i<n;i++){

        cout<<i<<"\t"
            <<AT[i]<<"\t"
            <<BT[i]<<"\t"
            <<CT[i]<<"\t"
            <<TAT[i]<<"\t"
            <<WT[i]<<"\t"
            <<RT[i]<<endl;
    }

    return 0;
}
