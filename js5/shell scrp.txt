LOOPS-loops.sh
#!/bin/bash
#for loop
echo "for loop eg: "
for i in 1 2 3 4
do
 echo "item is: $i"
done

#whole loop
echo "while loop eg:"
i=1
while [ $i -lt 5 ]
do
    echo "item is: $i"
    i=` expr $i + 1 `
done

#until loop
echo "until loop eg: "
i=0
until [ ! $i -lt 5 ]
do 
    echo "item is: $i"
    i=` expr $i + 1 `
done

#nested loop
echo "nested loop eg: "
for i in 1 2 3
do 
 j=$i
 while [ $j -gt 0 ]
 do
  echo -n "x"
  j=` expr $j - 1 `
  done
  echo #move to new line now
done

#break and continue
for i in 1 2 3 4 5
do
    if [ $i -eq 3 ]
    then
        break
    fi

    echo $i
done

for i in 1 2 3 4 5
do
    if [ $i -eq 3 ]
    then
        continue
    fi

    echo $i
done
